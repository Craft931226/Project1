    if (buf[i] == 0x90)                       exit_with("no NOP sled!!!");      // 禁止 NOP (0x90)
    if (i <= 0x76 && buf[i]==0x0f && buf[i+1]==0x05)
                                               exit_with("no syscall!!!");      // 禁止 0f 05 (syscall)

                                               exit_with("no int 0x80!!!");     // 禁止 cd 80 (int 0x80)
}

    if (!(buf[idx] == 0x0c && buf[idx+1] == 0x87))
